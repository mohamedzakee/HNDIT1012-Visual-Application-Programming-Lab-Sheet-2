Setup 10 min

  1. Open Visual Studio 2022 → Create a new project.
  2. Search Windows Forms and choose Windows Forms App (.NET Framework) with the C# tag → Next.
  3. Project name: CNSLab · Framework: .NET Framework 4.8 (4.7.2 is also fine) → Create.
  4. Click on the form. In the Properties window (F4) set: Text = CNS Lab – your Reg No · Size = 650, 480 · Font = Segoe UI, 10pt.
  5. From the Toolbox (Containers group), drag a TabControl onto the form. Set Dock = Fill.
  6. Select the TabControl → Properties → TabPages → click …. Click Add until there are 7 pages, and set each page’s Text:
    Ex1 Data · Ex2 Generations · Ex3 Units · Ex4 Numbers · Ex5 ASCII · Ex6 BCD · Ex7 Gates
  7. Press F5. You should see a window with 7 tabs. Close it.
Two important rules ① Choose the template with (.NET Framework) in its name, not the plain “Windows Forms App”.
② Always set a control’s (Name) before you double-click it. The code uses these exact names. Capital letters matter.

private void btnProcess_Click(object sender, EventArgs e)
{
    // DATA: raw marks typed by the user
    int m1 = int.Parse(txtM1.Text);
    int m2 = int.Parse(txtM2.Text);
    int m3 = int.Parse(txtM3.Text);
    int m4 = int.Parse(txtM4.Text);
    int m5 = int.Parse(txtM5.Text);

    // PROCESSING
    int total = m1 + m2 + m3 + m4 + m5;
    double average = total / 5.0;

    string grade;
    if (average >= 75) grade = "A";
    else if (average >= 65) grade = "B";
    else if (average >= 50) grade = "C";
    else if (average >= 35) grade = "S";
    else grade = "F";

    // INFORMATION: meaningful output
    lblTotal.Text = "Total: " + total;
    lblAverage.Text = "Average: " + average.ToString("0.00");
    lblGrade.Text = "Grade: " + grade;
}

Form1.cs
private void btnBcd_Click(object sender, EventArgs e)
{
    string bcd = "";

    // take one digit at a time
    foreach (char c in txtBcd.Text)
    {
        int digit = c - '0';    // '5' becomes 5
        bcd = bcd + Convert.ToString(digit, 2).PadLeft(4, '0') + " ";
    }

    lblBcd.Text = "BCD: " + bcd;
    lblPureBinary.Text = "Pure binary: " + Convert.ToString(int.Parse(txtBcd.Text), 2);
}

C# logic operators && = AND · || = OR · ! = NOT · ^ = XOR. Convert.ToInt32(true) gives 1 and Convert.ToInt32(false) gives 0.

② Code: double-click btnGates

Form1.cs
private void btnGates_Click(object sender, EventArgs e)
{
    bool a = chkA.Checked;    // ticked = true = 1
    bool b = chkB.Checked;

    // The 7 logic gates
    lblAnd.Text  = "AND   (A · B)    = " + Convert.ToInt32(a && b);
    lblOr.Text   = "OR    (A + B)    = " + Convert.ToInt32(a || b);
    lblNot.Text  = "NOT   (A')       = " + Convert.ToInt32(!a);
    lblNand.Text = "NAND  ((A · B)') = " + Convert.ToInt32(!(a && b));
    lblNor.Text  = "NOR   ((A + B)') = " + Convert.ToInt32(!(a || b));
    lblXor.Text  = "XOR   (A ⊕ B)    = " + Convert.ToInt32(a ^ b);
    lblXnor.Text = "XNOR  ((A ⊕ B)') = " + Convert.ToInt32(!(a ^ b));

    // De Morgan's theorem: left side must always equal right side
    lblDeMorgan1.Text = "(A · B)' = " + Convert.ToInt32(!(a && b))
                      + "     A' + B' = " + Convert.ToInt32(!a || !b);
    lblDeMorgan2.Text = "(A + B)' = " + Convert.ToInt32(!(a || b))
                      + "     A' · B' = " + Convert.ToInt32(!a && !b);
}
